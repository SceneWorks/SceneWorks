+/// first read's tail is exactly zero — the signature of an ignored short read.
+fn load_audit(path: &Path, tensors: &HashMap<String, Array>) {
+    let Ok(log) = std::env::var("SCENEWORKS_LTX25_LOAD_AUDIT") else {
+        return;
+    };
+    let floor: usize = std::env::var("SCENEWORKS_LTX25_LOAD_AUDIT_MIN")
+        .ok()
+        .and_then(|value| value.parse().ok())
+        .unwrap_or(64 * 1024 * 1024);
+    let mut names: Vec<&String> = tensors
+        .keys()
+        .filter(|name| tensors[*name].nbytes() >= floor)
+        .collect();
+    if names.is_empty() {
+        return;
+    }
+    names.sort();
+    let first: Vec<Option<Vec<f32>>> = names.iter().map(|n| host_f32(&tensors[*n])).collect();
+    let reload = Array::load_safetensors(path).ok();
+    let mut out = String::new();
+    for (index, name) in names.iter().enumerate() {
+        let second = reload
+            .as_ref()
+            .and_then(|map| map.get(*name))
+            .and_then(host_f32);
+        let line = match (&first[index], &second) {
+            (Some(a), Some(b)) if a == b => {
+                format!("{} {name} bytes={} MATCH fnv={:016x}", path.display(), tensors[*name].nbytes(), fnv(a))
+            }
+            (Some(a), Some(b)) => {
+                let first_diff = a.iter().zip(b).position(|(x, y)| x != y).unwrap_or(a.len());
+                let zero_tail = a.iter().rev().take_while(|v| **v == 0.0).count();
+                let zeros_total = a.iter().filter(|v| **v == 0.0).count();
+                format!(
+                    "{} {name} bytes={} MISMATCH elems={} first_diff_elem={} zero_tail_elems={} zeros_in_first={} zeros_in_second={} fnv_first={:016x} fnv_second={:016x}",
+                    path.display(), tensors[*name].nbytes(), a.len(), first_diff, zero_tail,
+                    zeros_total, b.iter().filter(|v| **v == 0.0).count(), fnv(a), fnv(b)
+                )
+            }
+            _ => format!("{} {name} HASH-ERROR", path.display()),
+        };
+        out.push_str(&line);

+    }

+fn load_audit(path: &Path, tensors: &HashMap<String, Array>) {
+    let Ok(log) = std::env::var("SCENEWORKS_LTX25_LOAD_AUDIT") else {
+        return;
+    };
+    let floor: usize = std::env::var("SCENEWORKS_LTX25_LOAD_AUDIT_MIN")
+        .ok()
+        .and_then(|value| value.parse().ok())
+        .unwrap_or(64 * 1024 * 1024);
+    let mut names: Vec<&String> = tensors
+        .keys()
+        .filter(|name| tensors[*name].nbytes() >= floor)
+        .collect();
+    if names.is_empty() {
+        return;
+    }
+    names.sort();
+    // Hash the vulnerable first materializations before anything re-reads the file.
+    let first: Vec<Option<u64>> = names
+        .iter()
+        .map(|name| audit_hash(&tensors[*name]))
+        .collect();
+    let reload = Array::load_safetensors_with_metadata(path).ok();
+    let mut out = String::new();
+    for (index, name) in names.iter().enumerate() {
+        let second = reload
+            .as_ref()
+            .and_then(|(map, _)| map.get(*name))
+            .and_then(audit_hash);
+        let verdict = match (first[index], second) {
+            (Some(a), Some(b)) if a == b => "MATCH",
+            (Some(_), Some(_)) => "MISMATCH",
+            _ => "HASH-ERROR",
+        };
+        out.push_str(&format!(
+            "{} {} bytes={} first={:016x} second={:016x} {}\n",
+            path.display(),
+            name,
+            tensors[*name].nbytes(),
+            first[index].unwrap_or(0),
+            second.unwrap_or(0),
+            verdict,
+        ));
+    }
+    if let Ok(mut file) = std::fs::OpenOptions::new().create(true).append(true).open(&log) {
+        use std::io::Write;
+        let _ = file.write_all(out.as_bytes());
+    }
+}
+
+fn audit_hash(array: &Array) -> Option<u64> {
+    // CPU stream on purpose: a GPU cast of every cold tensor at once tripped the Metal watchdog
+    // (SubmissionsIgnored, 2026-09-01) and confounded the audit; this keeps the probe off the GPU.
+    let as_f32 = array
+        .as_dtype_device(Dtype::Float32, mlx_rs::StreamOrDevice::cpu())
+        .ok()?;
+    mlx_rs::transforms::eval([&as_f32]).ok()?;
+    let values = as_f32.as_slice::<f32>();
+    let mut hash: u64 = 0xcbf29ce484222325;
+    for value in values {
+        for byte in value.to_bits().to_le_bytes() {
+            hash ^= u64::from(byte);
+            hash = hash.wrapping_mul(0x0000_0100_0000_01b3);
+        }
+    }
+    Some(hash)
+}
