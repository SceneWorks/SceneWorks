    let mut held = Vec::new();

    while left > 0 {
        let chunk = left.min(2);
        held.push(Tensor::zeros((chunk << 30,), DType::U8, &device).expect("alloc"));
        left -= chunk;

        std::thread::sleep(std::time::Duration::from_secs(3600));
    }
}
